- Traversing the Linux TCP/IP network stack adds $8\text{ }\mu\text{s} - 25\text{ }\mu\text{s}$ of latency.

    input  logic signed [31:0]       rag_signal_fp,     // Raw LLM RAG conviction [-1.0, +1.0]
    input  logic signed [31:0]       delta_i_rag_fp,    // KL Surprise D_KL(P_rag || P_prior)
    input  logic                     c2pa_root_signed,  // 1 = Verified SEC EDGAR / Wire C2PA Root

    input  logic signed [31:0]       weighted_bid_depth_fp,
    input  logic signed [31:0]       weighted_ask_depth_fp,
    input  logic signed [31:0]       e_capital_score_fp,    // Combined VPIN + Kyle Lambda + Opt Skew
    input  logic signed [31:0]       pre_news_wall_ratio_fp,// Fraction of wall resting prior to news
    input  logic signed [31:0]       deriv_divergence_fp,   // (1.0 - Sigmoid(Options_Skew_Z))

    // Deterministic Outbound Execution Interface (To OUCH 5.0 / FIX MAC TX)
    output logic                     tx_order_enable,       // 1 = Pass to Exchange, 0 = KILLED
    output logic                     sec_15c3_8_halt_irq,   // Triggers 15s Epistemic Halt & CAT Log
    output logic [63:0]              order_id_out,
    output logic signed [31:0]       approved_signal_fp,
    output logic [3:0]               rejection_reason_code
);

    // Rejection Reason Telemetry Codes for SEC Consolidated Audit Trail (CAT)
    localparam logic [3:0] CODE_VERIFIED_PASS         = 4'h0;
    localparam logic [3:0] CODE_C2PA_EXEMPT_PASS      = 4'h1;
    localparam logic [3:0] CODE_PREDATORY_ASK_TRAP    = 4'hE;
    localparam logic [3:0] CODE_PREDATORY_BID_TRAP    = 4'hD;
    localparam logic [3:0] CODE_DATA_VOID_UNBACKED_KL = 4'hF;

    logic [63:0]              s1_order_id, s2_order_id;
    logic                     s1_c2pa, s2_c2pa;
    logic signed [31:0]       s1_rag_sig, s2_rag_sig;
    logic signed [31:0]       s1_prior_sig, s2_prior_sig;
    logic signed [31:0]       s1_delta_i, s2_delta_i;

    logic signed [63:0]       prod_cap_Allowance;
    logic signed [31:0]       s1_allowed_surprise_fp;
    logic signed [31:0]       s1_depth_diff_fp;
    logic signed [31:0]       s1_depth_sum_fp;
    logic signed [63:0]       prod_wall_deriv;
    logic signed [31:0]       s1_wall_factor_fp;

    logic signed [63:0]       prod_trap_thresh;
    logic signed [31:0]       s2_excess_surprise_fp;

    logic                     s2_bearish_bid_trap;

    // PIPELINE STAGE 1: LOB ASYMMETRY & CAPITAL CAPACITY MULTIPLICATION
    // =========================================================================
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            s1_valid               <= 1'b0;
            s1_order_id            <= 64'd0;
            s1_c2pa                <= 1'b0;
            s1_rag_sig             <= 32'sd0;
            s1_prior_sig           <= 32'sd0;
            s1_delta_i             <= 32'sd0;
            s1_allowed_surprise_fp <= 32'sd0;
            s1_depth_diff_fp       <= 32'sd0;
            s1_depth_sum_fp        <= 32'sd1;
            s1_wall_factor_fp      <= 32'sd0;

            s1_valid      <= order_valid_in;
            s1_order_id   <= order_id_in;
            s1_c2pa       <= c2pa_root_signed;
            s1_rag_sig    <= rag_signal_fp;

            prod_cap_Allowance     = (64'(KAPPA_CAP_FP) * 64'(e_capital_score_fp)) >>> FRAC_BITS;
            s1_allowed_surprise_fp <= prod_cap_Allowance[31:0];

            // Compute Order Book Asymmetry Numerator (Ask - Bid) and Denominator (Ask + Bid)
            s1_depth_diff_fp <= weighted_ask_depth_fp - weighted_bid_depth_fp;
            s1_depth_sum_fp  <= (weighted_ask_depth_fp + weighted_bid_depth_fp > 32'sd0)
                                ? (weighted_ask_depth_fp + weighted_bid_depth_fp) : 32'sd1;

            // Pre-multiply 2.0 * pre_news_wall_ratio * deriv_divergence (Division-free OBAR check)
            prod_wall_deriv   = (64'(pre_news_wall_ratio_fp) * 64'(deriv_divergence_fp)) >>> (FRAC_BITS - 1);
            s1_wall_factor_fp <= prod_wall_deriv[31:0];
        end
    end

    // PIPELINE STAGE 2: DIVISION-FREE PREDATORY TRAP & SURPRISE COMPARISON
    // Note: Avoids slow hardware division by cross-multiplying denominator:
    //       (|Ask - Bid| / (Ask + Bid)) * WallFactor >= TAU_TRAP
    //  <==> |Ask - Bid| * WallFactor >= TAU_TRAP * (Ask + Bid)

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            s2_valid              <= 1'b0;
            s2_order_id           <= 64'd0;
            s2_c2pa               <= 1'b0;
            s2_rag_sig            <= 32'sd0;
            s2_prior_sig          <= 32'sd0;
            s2_excess_surprise_fp <= 32'sd0;
            s2_bullish_ask_trap   <= 1'b0;
            s2_bearish_bid_trap   <= 1'b0;
        end else begin
            s2_valid     <= s1_valid;
            s2_order_id  <= s1_order_id;
            s2_c2pa      <= s1_c2pa;
            s2_rag_sig   <= s1_rag_sig;
            s2_prior_sig <= s1_prior_sig;

            // Excess informational surprise unbacked by LOB capital energy
            if (s1_delta_i > s1_allowed_surprise_fp)
                s2_excess_surprise_fp <= s1_delta_i - s1_allowed_surprise_fp;
            else
                s2_excess_surprise_fp <= 32'sd0;

            // Cross-multiplied Trap Condition (Zero hardware divider latency!)
            prod_trap_thresh = (64'(TAU_TRAP_FP) * 64'(s1_depth_sum_fp)) >>> FRAC_BITS;

            // Case A: RAG screams BUY (+), Ask book has massive pre-positioned wall (Ask > Bid)
            if (s1_rag_sig > RAG_SIG_THRESH && s1_depth_diff_fp > 32'sd0) begin
                prod_trap_lhs = (64'(s1_depth_diff_fp) * 64'(s1_wall_factor_fp)) >>> FRAC_BITS;
                s2_bullish_ask_trap <= (prod_trap_lhs >= prod_trap_thresh);
            end else begin
                s2_bullish_ask_trap <= 1'b0;
            end

            // Case B: RAG screams SELL (-), Bid book has massive pre-positioned wall (Bid > Ask)
            if (s1_rag_sig < -RAG_SIG_THRESH && s1_depth_diff_fp < 32'sd0) begin
                prod_trap_lhs = (64'(-s1_depth_diff_fp) * 64'(s1_wall_factor_fp)) >>> FRAC_BITS;
                s2_bearish_bid_trap <= (prod_trap_lhs >= prod_trap_thresh);
            end else begin
                s2_bearish_bid_trap <= 1'b0;
            end
        end
    end

    // =========================================================================
    // PIPELINE STAGE 3: DETERMINISTIC SEC RULE 15c3-8 KILL-SWITCH OUTPUT GATE
    // =========================================================================
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            tx_order_enable       <= 1'b0;
            sec_15c3_8_halt_irq   <= 1'b0;
            order_id_out          <= 64'd0;
            approved_signal_fp    <= 32'sd0;
            rejection_reason_code <= CODE_VERIFIED_PASS;
        end else if (!s2_valid) begin
            tx_order_enable       <= 1'b0;
            sec_15c3_8_halt_irq   <= 1'b0;
        end else begin
            order_id_out <= s2_order_id;

                approved_signal_fp    <= s2_prior_sig;    // Revert to safe prior
                rejection_reason_code <= CODE_PREDATORY_ASK_TRAP;
            end
            else if (s2_bearish_bid_trap) begin
                tx_order_enable       <= 1'b0;
                sec_15c3_8_halt_irq   <= 1'b1;
                approved_signal_fp    <= s2_prior_sig;
                rejection_reason_code <= CODE_PREDATORY_BID_TRAP;
            end
            // Gate 2: Unanchored Breaking-News Data-Void Spoof (High KL Surprise, Zero LOB Energy)

                tx_order_enable       <= 1'b0;
                sec_15c3_8_halt_irq   <= 1'b1;
                approved_signal_fp    <= s2_prior_sig;
                rejection_reason_code <= CODE_DATA_VOID_UNBACKED_KL;
            end
            // Gate 3: Verified Order (Either C2PA Root-Signed or Confirmed by LOB Energy)
            else begin
                tx_order_enable       <= 1'b1;
                sec_15c3_8_halt_irq   <= 1'b0;
                approved_signal_fp    <= s2_rag_sig;
                rejection_reason_code <= s2_c2pa ? CODE_C2PA_EXEMPT_PASS : CODE_VERIFIED_PASS;
            end
        end
    end
